"""
accident_simulation.py
──────────────────────
Runs SUMO via TraCI. Injects 2 random accidents per simulation.
Lane changing is handled entirely by SUMO's default LC2013 model.
Accidents stop a vehicle in place; surrounding vehicles slow down
and change lanes / queue naturally via SUMO's built-in behavior.

Usage:
    python accident_simulation.py           # GUI, 2 accidents
    python accident_simulation.py --nogui   # headless
    python accident_simulation.py --seed 42 # reproducible
"""

import sys
import random
import argparse
import traci

# ── Argument parsing ───────────────────────────────────────────────────────
parser = argparse.ArgumentParser(description="SUMO accident simulation")
parser.add_argument("--nogui", action="store_true")
parser.add_argument("--seed",  type=int, default=None)
args = parser.parse_args()

if args.seed is not None:
    random.seed(args.seed)

# ── SUMO binary ────────────────────────────────────────────────────────────
SUMO_BINARY = "sumo" if args.nogui else "sumo-gui"
SUMO_CMD = [
    SUMO_BINARY,
    "-c", "intersection2.sumocfg",
    "--no-step-log",
    "--waiting-time-memory", "1000",
]

# ── Configuration ──────────────────────────────────────────────────────────
SIM_END           = 3600
NUM_ACCIDENTS     = 2          # exactly 2 per simulation run
MIN_DURATION      = 60         # shortest blockage (seconds)
MAX_DURATION      = 180        # longest blockage (seconds)
SLOWDOWN_RADIUS   = 50.0       # rubbernecking slow zone (metres)
SLOWDOWN_FACTOR   = 0.2        # reduce speed to 20% of normal
EARLIEST_ACCIDENT = 300        # don't trigger before t=300s
LATEST_ACCIDENT   = SIM_END - MAX_DURATION - 60

# ── Accident state ─────────────────────────────────────────────────────────
class Accident:
    def __init__(self, trigger_time):
        self.trigger_time = trigger_time
        self.vehicle_id   = None
        self.end_time     = None
        self.position     = None
        self.slowed       = {}     # {veh_id: original_maxSpeed}
        self.active       = False
        self.cleared      = False

def schedule_accidents(n):
    # Pick n non-overlapping random trigger times spread across the day
    times = sorted(random.sample(range(EARLIEST_ACCIDENT, LATEST_ACCIDENT), n))
    return [Accident(t) for t in times]

# ── Geometry helper ────────────────────────────────────────────────────────
def distance(p1, p2):
    return ((p1[0]-p2[0])**2 + (p1[1]-p2[1])**2) ** 0.5

# ── Activate accident ──────────────────────────────────────────────────────
def activate_accident(accident, step):
    vehicles = traci.vehicle.getIDList()
    if not vehicles:
        print(f"  [t={step}] No vehicles on road, retrying in 30s.")
        return False

    victim   = random.choice(vehicles)
    duration = random.randint(MIN_DURATION, MAX_DURATION)
    lane_id  = traci.vehicle.getLaneID(victim)
    lane_pos = traci.vehicle.getLanePosition(victim)
    x, y     = traci.vehicle.getPosition(victim)
    edge_id  = traci.vehicle.getRoadID(victim)

    # Freeze the victim — let SUMO handle everything behind it
    traci.vehicle.setSpeed(victim, 0)
    traci.vehicle.setSpeedMode(victim, 0)        # ignore safety checks on victim
    # NOTE: lane-change mode is NOT overridden for other vehicles
    # SUMO's LC2013 model handles all lane changes automatically
    traci.vehicle.setColor(victim, (255, 0, 0, 255))   # RED

    accident.vehicle_id = victim
    accident.end_time   = step + duration
    accident.position   = (x, y)
    accident.active     = True

    print(f"\n🚨 ACCIDENT at t={step}s")
    print(f"   Vehicle  : {victim}")
    print(f"   Edge     : {edge_id}  Lane: {lane_id}")
    print(f"   Position : x={x:.1f}, y={y:.1f}  lane_pos={lane_pos:.1f}m")
    print(f"   Duration : {duration}s  →  clears at t={accident.end_time}s")
    print(f"   ↳ SUMO LC2013 model handles lane changes automatically")
    return True

# ── Rubbernecking slowdown ─────────────────────────────────────────────────
def apply_slowdown(accident):
    ax, ay = accident.position
    for vid in traci.vehicle.getIDList():
        if vid == accident.vehicle_id or vid in accident.slowed:
            continue
        vx, vy = traci.vehicle.getPosition(vid)
        if distance((vx, vy), (ax, ay)) <= SLOWDOWN_RADIUS:
            original = traci.vehicle.getMaxSpeed(vid)
            traci.vehicle.setMaxSpeed(vid, original * SLOWDOWN_FACTOR)
            traci.vehicle.setColor(vid, (0, 200, 255, 255))  # CYAN
            accident.slowed[vid] = original

# ── Restore after clearance ────────────────────────────────────────────────
def restore_all(accident):
    for vid, spd in accident.slowed.items():
        try:
            traci.vehicle.setMaxSpeed(vid, spd)
            traci.vehicle.setSpeedMode(vid, 31)   # restore default speed mode
            traci.vehicle.setColor(vid, (255, 255, 0, 255))  # YELLOW
        except traci.exceptions.TraCIException:
            pass  # vehicle already left the simulation
    accident.slowed.clear()

def clear_accident(accident, step):
    print(f"\n✅ CLEARED  at t={step}s  —  {accident.vehicle_id}")
    restore_all(accident)
    try:
        traci.vehicle.remove(accident.vehicle_id)
    except traci.exceptions.TraCIException:
        pass
    accident.active  = False
    accident.cleared = True

# ── Main simulation loop ───────────────────────────────────────────────────
def run():
    traci.start(SUMO_CMD)
    accidents = schedule_accidents(NUM_ACCIDENTS)

    print(f"\n{'='*55}")
    print(f"  Accidents scheduled at: {[a.trigger_time for a in accidents]}")
    print(f"  Lane changes : SUMO default LC2013 model")
    print(f"{'='*55}\n")

    step = 0
    while step < SIM_END:
        traci.simulationStep()

        for accident in accidents:

            # Trigger
            if (not accident.active and not accident.cleared
                    and step >= accident.trigger_time):
                if not activate_accident(accident, step):
                    accident.trigger_time = step + 30

            # Active — keep frozen + apply slowdown
            if accident.active:
                try:
                    traci.vehicle.setSpeed(accident.vehicle_id, 0)
                except traci.exceptions.TraCIException:
                    clear_accident(accident, step)
                    continue

                apply_slowdown(accident)

                # Clear when duration elapsed
                if step >= accident.end_time:
                    clear_accident(accident, step)

        step += 1

    print(f"\n{'='*55}")
    print(f"  Simulation complete — {NUM_ACCIDENTS} accidents simulated")
    print(f"{'='*55}\n")
    traci.close()
    sys.stdout.flush()

if __name__ == "__main__":
    run()
